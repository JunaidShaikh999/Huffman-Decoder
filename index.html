// Read URL parameters const urlParams = new URLSearchParams(window.location.search); const bits = urlParams.get('bits'); const keyParam = urlParams.get('key'); if (bits && keyParam) { // Reconstruct the Huffman map from "ASCII:code|ASCII:code|" const reverseDict = {}; const pairs = keyParam.split('|').filter(Boolean); pairs.forEach(pair => { const [charCode, code] = pair.split(':'); const symbol = String.fromCharCode(parseInt(charCode, 10)); reverseDict[code] = symbol; // Binary Code -> Character }); // Decode the bitstring let decodedText = ''; let currentCode = ''; for (let bit of bits) { currentCode += bit; if (reverseDict[currentCode]) { decodedText += reverseDict[currentCode]; currentCode = ''; } } document.getElementById('result').innerText = decodedText; } else { document.getElementById('result').innerText = "No Huffman data found in link."; }
